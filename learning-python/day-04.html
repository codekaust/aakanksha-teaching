The program

01def count_characters(file_name):
02    vowel = "aeiouAEIOU"
03    vowel_count = 0
04    consonant_count = 0
05    uppercase_count = 0
06    lowercase_count = 0
07    with open(file_name, "r") as file:
08        text = file.read()
09    for char in text:
10        if char.isalpha():
11            if char in vowel:
12                vowel_count += 1
13            else:
14                consonant_count += 1
15            if char.isupper():
16                uppercase_count += 1
17            elif char.islower():
18                lowercase_count += 1
19    print("No. of vowels =", vowel_count)
20    print("No. of consonants =", consonant_count)
21    print("No. of uppercase =", uppercase_count)
22    print("No. of lowercase =", lowercase_count)
23def create_text_file(file_name, content):
24    with open(file_name, "w") as file:
25        file.write(content)
26file_name = "letters.txt"
27content = input("Enter a sentence: ")
28create_text_file(file_name, content)
29count_characters(file_name)
